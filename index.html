        /* ==========================================================================

        class SoundEngine {

            constructor() {

                this.ctx = null;

                this.enabled = true;

            }


            init() {

                if (!this.ctx) {

                    const AudioContext = window.AudioContext || window.webkitAudioContext;

                    if (AudioContext) {

                        this.ctx = new AudioContext();

                    }

                }

                if (this.ctx && this.ctx.state === 'suspended') {

                    this.ctx.resume();

                }

            }


            playJump() {

                if (!this.enabled) return;

                this.init();

                if (!this.ctx) return;


                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();


                osc.type = 'square';

                osc.frequency.setValueAtTime(150, now);

                osc.frequency.exponentialRampToValueAtTime(450, now + 0.15);


                gain.gain.setValueAtTime(0.12, now);

                gain.gain.linearRampToValueAtTime(0.01, now + 0.15);


                osc.connect(gain);

                gain.connect(this.ctx.destination);


                osc.start(now);

                osc.stop(now + 0.15);

            }


            playCoin() {

                if (!this.enabled) return;

                this.init();

                if (!this.ctx) return;


                const now = this.ctx.currentTime;

                const osc = this.ctx.createOscillator();

                const gain = this.ctx.createGain();


                osc.type = 'sine';

                osc.frequency.setValueAtTime(987.77, now); // B5

                osc.frequency.setValueAtTime(1318.51, now + 0.08); // E6


                gain.gain.setValueAtTime(0.15, now);

                gain.gain.linearRampToValueAtTime(0.01, now + 0.2);


                osc.connect(gain);

                gain.connect(this.ctx.destination);


                osc.start(now);

                osc.stop(now + 0.2);

            }


            playPowerup() {

                if (!this.enabled) return;

                this.init();

                if (!this.ctx) return;


                const now = this.ctx.currentTime;

                const notes = [261.63, 329.63, 392.00, 523.25]; // C4, E4, G4, C5

                notes.forEach((freq, idx) => {

                    const osc = this.ctx.createOscillator();

                    const gain = this.ctx.createGain();

                    osc.type = 'triangle';

                    osc.frequency.setValueAtTime(freq, now + idx * 0.06);


                    gain.gain.setValueAtTime(0.15, now + idx * 0.06);

                    gain.gain.exponentialRampToValueAtTime(0.01, now + idx * 0.06 + 0.12);


                    osc.connect(gain);

                    gain.connect(this.ctx.destination);


                    osc.start(now + idx * 0.06);

                    osc.stop(now + idx * 0.06 + 0.12);

                });

            }

                if (this.powerups.boots > 0) this.powerups.boots--;

        function gameLoop(timestamp) {

            if (!gameRunning) return;


            if (!gamePaused) {

                updateGame();

                renderGame();

            }


            animationFrameId = requestAnimationFrame(gameLoop);

        }


        function updateGame() {

            // Speed scaling over distance

            distanceMeters += Math.floor(currentSpeed * 0.1);

            currentSpeed = baseSpeed + (distanceMeters * 0.0015);


            // Speed boots active boost

            const effectiveSpeed = miner.powerups.boots > 0 ? currentSpeed * 1.35 : currentSpeed;


            // Boulder distance recovery when running cleanly

            if (boulderOffset < MAX_BOULDER_DIST) {

                boulderOffset += 0.15;

            }


            // Update Miner & Boulder

            miner.update();

            cryptoBoulder.update(miner.x, boulderOffset);


            // Check if Boulder crushed Miner

            if (boulderOffset <= MIN_BOULDER_DIST) {

                triggerGameOver();

                return;

            }


            // Update Parallax Background

            bgLayers.forEach(layer => {

                layer.x -= effectiveSpeed * layer.speedMult;

                if (layer.x <= -V_WIDTH) layer.x = 0;

            });


            // Update Obstacles

            for (let i = obstacles.length - 1; i >= 0; i--) {

                const obs = obstacles[i];

                obs.x -= effectiveSpeed;


                // Collision Detection with Miner

                if (checkCollision(miner, obs)) {

                    if (miner.powerups.pickaxe > 0 && obs.destructible) {

                        // Golden Pickaxe Smashes Rock!

                        createExplosion(obs.x + obs.width / 2, obs.y + obs.height / 2, '#fcc419', 16);

                        soundEngine.playSmash();

                        obstacles.splice(i, 1);

                        continue;

                    } else if (miner.invulnerableTimer === 0) {

                        // Stumble / Hit Obstacle!

                        soundEngine.playHit();

                        createExplosion(miner.x + 10, miner.y + 10, '#ff6b6b', 10);

                        miner.invulnerableTimer = 50; // Invulnerability frames

                        boulderOffset -= 75; // Boulder gains on miner!


                        // Camera Shake Effect on UI

                        document.querySelector('main').classList.add('shake');

                        setTimeout(() => document.querySelector('main').classList.remove('shake'), 400);


                        if (obs.type === 'lava_pit') {

                            boulderOffset -= 40; // Heavy penalty for lava pit

                        }

                    }

                }


                // Remove off-screen obstacles

                if (obs.x + obs.width < -50) {

                    obstacles.splice(i, 1);

                }

            }


            // Update Collectibles

            for (let i = collectibles.length - 1; i >= 0; i--) {

                const item = collectibles[i];

                item.x -= effectiveSpeed;


                // Check Pickup

                const dist = Math.hypot((miner.x + miner.width / 2) - item.x, (miner.y + miner.height / 2) - item.y);

                if (dist < miner.width / 2 + item.radius) {

                    if (item.type === 'coin') {

                        coinsCollected++;

                        soundEngine.playCoin();

                        createExplosion(item.x, item.y, '#fcc419', 6);

                    } else if (item.type === 'powerup_pickaxe') {

                        miner.powerups.pickaxe = 300; // ~5 sec duration

                        soundEngine.playPowerup();

                        createExplosion(item.x, item.y, '#fcc419', 14);

                    } else if (item.type === 'powerup_boots') {

                        miner.powerups.boots = 300; // ~5 sec duration

                        soundEngine.playPowerup();

                        createExplosion(item.x, item.y, '#22d3ee', 14);

                    }

                    collectibles.splice(i, 1);

                    continue;

                }


                if (item.x < -30) collectibles.splice(i, 1);

            }


            // Update Particles

            for (let i = particles.length - 1; i >= 0; i--) {

                const p = particles[i];

                p.x += p.vx;

                p.y += p.vy;

                p.life--;

                if (p.life <= 0) particles.splice(i, 1);

            }


            // Spawn fresh entities

            spawnEntities();


            // Update HUD

            document.getElementById('hud-coins').innerText = coinsCollected;

            document.getElementById('hud-distance').innerText = `${distanceMeters}m`;

            updatePowerupHUD();

        }

        function checkCollision(rect1, rect2) {

            return (

                rect1.x < rect2.x + rect2.width &&

                rect1.x + rect1.width > rect2.x &&

                rect1.y < rect2.y + rect2.height &&

                rect1.y + rect1.height > rect2.y

            );

        }


        function updatePowerupHUD() {

            const hud = document.getElementById('powerup-hud');

            hud.innerHTML = '';


            if (miner.powerups.pickaxe > 0) {

                const sec = (miner.powerups.pickaxe / 60).toFixed(1);

            if (miner.powerups.boots > 0) {

                const sec = (miner.powerups.boots / 60).toFixed(1);

                hud.innerHTML += `<div class="bg-pixel-cyan/90 border border-white text-black text-[10px] px-2 py-1 rounded font-mono font-bold flex items-center gap-1">⚡ BOOTS: ${sec}s</div>`;

            }

        }


        // KEYBOARD CONTROLS

        window.addEventListener('keydown', (e) => {

            if (!gameRunning || gamePaused) return;


            if (e.code === 'Space' || e.code === 'ArrowUp' || e.code === 'KeyW') {

                e.preventDefault();

                miner.jump();

            } else if (e.code === 'ArrowDown' || e.code === 'KeyS') {

                e.preventDefault();

                miner.slide();

            } else if (e.code === 'KeyP') {

                togglePause();

            }

        });


        // TOUCH CONTROLS (ON-SCREEN BUTTONS & CANVAS SWIPE)

        const btnJump = document.getElementById('btn-touch-jump');

        const btnSlide = document.getElementById('btn-touch-slide');


        btnJump.addEventListener('touchstart', (e) => {

            e.preventDefault();

            if (gameRunning && !gamePaused) miner.jump();

        });


        btnSlide.addEventListener('touchstart', (e) => {

            e.preventDefault();

            if (gameRunning && !gamePaused) miner.slide();

        });


        // Mouse click on canvas triggers jump

        canvas.addEventListener('mousedown', () => {

            if (gameRunning && !gamePaused) miner.jump();

        });


        function startGame() {

            document.getElementById('screen-start').classList.add('hidden');

            resetGameVars();

            gameRunning = true;

            gamePaused = false;

            soundEngine.init();

            requestAnimationFrame(gameLoop);

        }


        function restartGame() {

            document.getElementById('screen-gameover').classList.add('hidden');

            resetGameVars();

            gameRunning = true;

            gamePaused = false;

            requestAnimationFrame(gameLoop);

        }


        function resetGameVars() {

            distanceMeters = 0;

            coinsCollected = 0;

            currentSpeed = baseSpeed;

            boulderOffset = 180;

            spawnTimer = 0;

            obstacles = [];

            collectibles = [];

            particles = [];

            miner.reset();

            initBackground();

            document.getElementById('hud-highscore').innerText = `${highScore}m`;

        }


        function togglePause() {

            if (!gameRunning) return;

            gamePaused = !gamePaused;

            const pauseScreen = document.getElementById('screen-pause');

            const pauseIcon = document.getElementById('pause-icon');


            if (gamePaused) {

                pauseScreen.classList.remove('hidden');

                pauseIcon.className = "fa-solid fa-play";

            } else {

                pauseScreen.classList.add('hidden');

                pauseIcon.className = "fa-solid fa-pause";

            }

        }


        function resetGameToStart() {

            gameRunning = false;

            gamePaused = false;

            document.getElementById('screen-pause').classList.add('hidden');

            document.getElementById('screen-start').classList.remove('hidden');

        }


        function triggerGameOver() {

            gameRunning = false;

            soundEngine.playGameOver();


            // Save High Score

            if (distanceMeters > highScore) {

                highScore = distanceMeters;

                localStorage.setItem('crypto_escape_highscore', highScore.toString());

            }


            const totalScore = distanceMeters + (coinsCollected * 25);


            document.getElementById('go-distance').innerText = `${distanceMeters}m`;

            document.getElementById('go-coins').innerText = coinsCollected;

            document.getElementById('go-score').innerText = `${totalScore} pts`;

            document.getElementById('go-highscore').innerText = `${highScore}m`;


            document.getElementById('screen-gameover').classList.remove('hidden');

        }


        function toggleSound() {

            soundEngine.enabled = !soundEngine.enabled;

            const icon = document.getElementById('sound-icon');

            icon.className = soundEngine.enabled ? "fa-solid fa-volume-high" : "fa-solid fa-volume-xmark text-pixel-red";

        }


        function shareScore() {

            const text = `🎮 I survived ${distanceMeters}m in Crypto Escape before getting crushed by the Crypto Boulder! Can you beat my high score? ⛏️💥`;

            if (navigator.clipboard) {

                navigator.clipboard.writeText(text);

                alert("Score card copied to clipboard!");

            } else {

                alert(text);

            }

        }
